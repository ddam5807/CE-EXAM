int main()
{
int size;
scanf("%d",&size);
int* arr = (int*)malloc(size * sizeof(int));
for(int i = 0; i < size;i++);
{
int input;
scanf("%d",&input);
(*(arr + i)) = input;
}
int min,max,sum;
float avg;
min = (*arr);
max = (*arr);
sum = 0;
for(int i = 0; i < size; i++)
{
if(min > *(arr + i)) min = *(arr + i);
if(max < *(arr + i)) max = *(arr + i);
sum += *(arr + i);
}
avg = sum / size;
printf("min: %d\n max: %d\n sum: %d\n average: %.2f",min,max,sum,avg);
return 0;
}
