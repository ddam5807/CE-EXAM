char *my_strstr(const char *haystack, const char *needle)
{
if((*needle) == NULL) return haystack; //(*needle) == "")
if(strlen(haystack) < strlen(needle)) return NULL;
for(int i = 0; i < strlen(haystack) - strlen(needle); i++)//i <= strlen(haystack)
{
if(strncmp(haystack + i,needle,strlen(needle)) == 0) return (char)i; //우리는 값이 같아지는 지점의 포인터를 반환해야하므로 return (haystack + i); 
}
